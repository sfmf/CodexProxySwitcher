cmake_minimum_required(VERSION 3.20)
project(CodexProxySwitcher LANGUAGES CXX RC)
add_executable(CodexProxySwitcher WIN32 main.cpp app.rc)
target_compile_features(CodexProxySwitcher PRIVATE cxx_std_17)
target_link_libraries(CodexProxySwitcher PRIVATE ole32 uuid shell32 comdlg32 advapi32 gdi32 user32)
if(MSVC)
  set_property(TARGET CodexProxySwitcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(CodexProxySwitcher PRIVATE /utf-8 /W4 /O1)
  target_link_options(CodexProxySwitcher PRIVATE /OPT:REF /OPT:ICF)
else()
  target_compile_options(CodexProxySwitcher PRIVATE -Os -Wall -Wextra -ffunction-sections -fdata-sections)
  target_link_options(CodexProxySwitcher PRIVATE -static -static-libgcc -static-libstdc++ -municode -s -Wl,--gc-sections)
endif()
